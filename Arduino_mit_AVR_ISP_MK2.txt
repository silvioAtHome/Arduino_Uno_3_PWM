in .ardunio15/packages/arduino/hardware/avr/1.8.6/boards.txt  folgendes anfügen:

als Board "AVRISP mkII w/ ATmega328" einstellen und als Programmer: "AVRISP mk II" einstellen

atmega328bb.name=AVRISP mkII w/ ATmega328

#atmega328bb.upload.using=arduino:avrispmkii
#atmega328bb.upload.maximum_size=30720
#atmega328bb.upload.speed=57600

atmega328bb.upload.tool=avrdude
atmega328bb.upload.programmer=avrispmkII
atmega328bb.upload.maximum_size=30720
atmega328bb.upload.maximum_data_size=2048
atmega328bb.upload.speed=115200
atmega328bb.upload.network.endpoint_upload=/pgm/upload
atmega328bb.upload.network.endpoint_sync=/pgm/sync
atmega328bb.upload.network.sync_return=204:SYNC
atmega328bb.upload.network.endpoint_reset=/log/reset
atmega328bb.upload.network.port=80

atmega328bb.bootloader.low_fuses=0xE2
atmega328bb.bootloader.high_fuses=0xDA
atmega328bb.bootloader.extended_fuses=0x05
atmega328bb.bootloader.path=arduino:atmega
atmega328bb.bootloader.file=ATmegaBOOT_168_atmega328_pro_8MHz.hex
atmega328bb.bootloader.unlock_bits=0x3F
atmega328bb.bootloader.lock_bits=0x0F
atmega328bb.build.mcu=atmega328p
# oder 16 MHz hier?
atmega328bb.build.f_cpu=16000000L
atmega328bb.build.core=arduino
atmega328bb.build.variant=standard
atmega328bb.build.board=AVR_UNO
